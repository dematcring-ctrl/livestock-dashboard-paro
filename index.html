function loadAI() { return fetchCSV("ai").then(csv => { const rows = parseCSV(csv); const hIdx = findHeaderRow(rows, ["reportingmonth", "aicentre"]); if (hIdx < 0) throw new Error("AI sheet columns not recognized"); const H = rows[hIdx]; const C = { month: findCol(H, ["reportingmonth"]), year: findCol(H, ["year"]), gewog: findCol(H, ["gewog"]), centre: findCol(H, ["aicentre"]), semen: findCol(H, ["sementype"]), ai: findCol(H, ["noofai"]), male: findCol(H, ["maleprogeny"]), female: findCol(H, ["femaleprogeny"]) }; const records = []; const dataMonths = []; let curMonth = null; for (let i = hIdx + 1; i < rows.length; i++) { const r = rows[i]; const m = C.month >= 0 ? String(r[C.month] || "").trim() : ""; if (norm(m) === "total") continue; if (m) { if (!(norm(m) in MONTH_IDX)) continue; const y = C.year >= 0 ? num(r[C.year]) : null; curMonth = monthKey(m, y || ""); } if (!curMonth) continue; // Robust AI gewog parsing: // 1) use the actual Gewog column when present, // 2) fall back to the AI centre value, // 3) use existing group-to-gewog map when available, // 4) keep the row instead of dropping it. const gewogFromCell = C.gewog >= 0 ? String(r[C.gewog] || "").trim() : ""; const centreFromCell = C.centre >= 0 ? String(r[C.centre] || "").trim() : ""; const mappedFromCentre = GROUP_GEWOG[norm(centreFromCell)] || ""; const mappedFromGroup = GROUP_GEWOG[norm(centreFromCell + " " + (C.semen >= 0 ? String(r[C.semen] || "") : ""))] || ""; const gewog = (gewogFromCell || mappedFromCentre || mappedFromGroup || centreFromCell || "Unknown").trim(); const rec = { key: curMonth, gewog: gewog, centre: centreFromCell, semen: C.semen >= 0 ? String(r[C.semen] || "").trim() : "", ai: C.ai >= 0 ? num(r[C.ai]) : null, male: C.male >= 0 ? num(r[C.male]) : null, female: C.female >= 0 ? num(r[C.female]) : null }; records.push(rec); if ((rec.ai || 0) + (rec.male || 0) + (rec.female || 0) > 0 && !dataMonths.includes(curMonth)) { dataMonths.push(curMonth); } } ST.ai = { records: records, dataMonths: dataMonths }; }); }
